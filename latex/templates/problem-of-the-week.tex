% ---------------------------------------------------------------------------------
% Problem types. \ProblemType{<kind>}{<moves>} sets \Stipulation (the caption) and
% \HowItWorks (the left-hand explanation).
% ---------------------------------------------------------------------------------
\ExplSyntaxOn
\NewDocumentCommand \ProblemType { m m }
  {
    \str_case:nnF {#1}
      {
        {direct}
          {
            \Stipulation { White~to~play~and~mate~in~#2 \quad \textbf{\#\,#2} }
            \HowItWorks
              {
                \Lead{Direct~mate.}~White~moves~first~and~forces~checkmate~in~at~most~
                \Moves{#2},~however~Black~defends.
                \par
                The~first~move,~the~\emph{key},~must~work~against~every~defense.~
                Keys~are~usually~quiet:~not~a~check~or~a~capture,~often~a~move~that~
                seems~to~give~Black~more~freedom.
                \par
                Look~for~threats~White~could~make,~then~ask~how~each~Black~reply~
                fails.~There~is~only~one~key~that~works.
              }
          }
        {help}
          {
            \Stipulation { Helpmate~in~#2 \quad \textbf{h\#\,#2} }
            \HowItWorks
              {
                \Lead{Helpmate.}~Black~moves~first,~and~both~sides~cooperate~so~that~
                White's~\Ordinal{#2}~move~checkmates~Black.
                \par
                Neither~side~is~trying~to~win:~the~puzzle~is~to~find~the~one~
                sequence~of~\Moves{#2}~each~that~ends~in~mate.~Black~often~blocks~
                its~own~king,~and~each~piece~has~a~job~to~do.
                \par
                Work~backward:~picture~a~mating~position,~then~find~how~to~reach~it.
              }
          }
        {self}
          {
            \Stipulation { Selfmate~in~#2 \quad \textbf{s\#\,#2} }
            \HowItWorks
              {
                \Lead{Selfmate.}~White~moves~first~and~forces~\emph{Black}~to~give~
                checkmate~on~Black's~\Ordinal{#2}~move.~Black~resists~and~will~not~
                mate~unless~forced~to.
                \par
                White~must~leave~Black~no~choice:~every~legal~Black~move~in~the~
                end~has~to~mate~the~White~king.
                \par
                It~is~chess~in~reverse,~so~look~for~ways~to~take~away~Black's~
                harmless~moves.
              }
          }
        {study}
          {
            \str_if_eq:nnTF {#2} {draw}
              { \Stipulation { White~to~play~and~draw \quad \textbf{=} } }
              { \Stipulation { White~to~play~and~win \quad \textbf{+} } }
            \HowItWorks
              {
                \Lead{Endgame~study.}~White~moves~first~and~\str_if_eq:nnTF {#2} {draw}
                { holds~the~draw }{ wins }~against~Black's~best~defense.
                \par
                Unlike~a~mate~problem,~there~is~no~move~limit.~The~solution~is~the~
                main~line~up~to~a~position~whose~result~is~clear,~and~it~usually~
                contains~a~surprising~idea.
                \par
                Check~your~line~against~Black's~strongest~replies,~not~just~the~
                obvious~ones.
              }
          }
      }
      { \msg_error:nnn { potw } { unknown-type } {#1} }
  }
\msg_new:nnn { potw } { unknown-type }
  { Unknown~problem~type~'#1'.~Use~direct,~help,~self~or~study. }

% "1 move", "2 moves", ...
\NewDocumentCommand \Moves { m }
  { #1~\int_compare:nNnTF {#1} = {1} { move } { moves } }
% "1st", "2nd", "3rd", "4th", ...
\NewDocumentCommand \Ordinal { m }
  {
    #1
    \int_compare:nNnTF { \int_mod:nn {#1} {100} / 10 } = {1}
      { th }
      {
        \int_case:nnF { \int_mod:nn {#1} {10} }
          { {1} {st} {2} {nd} {3} {rd} }
          { th }
      }
  }
